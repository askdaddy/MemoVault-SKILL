- [ ] **Step 1: 创建断言库**

```bash
#!/usr/bin/env bash
# scripts/e2e/lib/assert.sh - e2e assertion helpers.
# Bash 3.2. No set -e. No emoji. Source this file; do not execute.

E2E_PASS="${E2E_PASS:-0}"
E2E_FAIL="${E2E_FAIL:-0}"

e2e_pass() {
  E2E_PASS=$((E2E_PASS + 1))
  printf 'PASS  %s\n' "$1"
}

e2e_fail() {
  E2E_FAIL=$((E2E_FAIL + 1))
  printf 'FAIL  %s — %s\n' "$1" "${2:-}"
}

assert_eq() {
  local expected="$1" actual="$2" msg="$3"
  if [ "$expected" = "$actual" ]; then
    e2e_pass "$msg"
  else
    e2e_fail "$msg" "expected='$expected' actual='$actual'"
  fi
}

assert_contains() {
  local haystack="$1" needle="$2" msg="$3"
  case "$haystack" in
    *"$needle"*) e2e_pass "$msg" ;;
    *) e2e_fail "$msg" "missing '$needle' in: $haystack" ;;
  esac
}

assert_file() {
  local path="$1" msg="$2"
  if [ -f "$path" ]; then
    e2e_pass "$msg"
  else
    e2e_fail "$msg" "not a file: $path"
  fi
}

assert_grep() {
  local pattern="$1" file="$2" msg="$3"
  if [ -f "$file" ] && grep -Eq -- "$pattern" "$file" 2>/dev/null; then
    e2e_pass "$msg"
  else
    e2e_fail "$msg" "pattern /$pattern/ not in $file"
  fi
}

# Run command; expect non-zero exit. Usage: assert_exit_nonzero "msg" cmd args...
assert_exit_nonzero() {
  local msg="$1"; shift
  local rc=0
  "$@" >/dev/null 2>&1 || rc=$?
  if [ "$rc" -ne 0 ]; then
    e2e_pass "$msg"
  else
    e2e_fail "$msg" "expected non-zero exit, got 0"
  fi
}

e2e_summary() {
  printf '\n=== e2e summary: pass=%s fail=%s ===\n' "$E2E_PASS" "$E2E_FAIL"
  [ "$E2E_FAIL" -eq 0 ]
}
```

  - `e2e_obsidian_json` → 打印 `obsidian.json` 路径或 return 1

- [ ] **Step 1: 写 `env.sh`**

```bash
#!/usr/bin/env bash
# scripts/e2e/lib/env.sh

e2e_resolve_repo() {
  local here
  here="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
  # BASH_SOURCE when sourced from lib/: lib -> e2e -> scripts; need repo root
  here="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
  E2E_REPO="$(cd "$here/../.." && pwd)"
  if [ -n "${E2E_REPO_OVERRIDE:-}" ]; then
    E2E_REPO="$(cd "$E2E_REPO_OVERRIDE" && pwd)"
  fi
  E2E_MM="$E2E_REPO/scripts/memovault.sh"
  [ -x "$E2E_MM" ] || { printf 'e2e: helper not executable: %s\n' "$E2E_MM" >&2; return 1; }
  E2E_STEM="E2E $(date +%s)"
}

e2e_setup_vault() {
  E2E_ROOT="$(mktemp -d "${TMPDIR:-/tmp}/memovault-e2e.XXXXXX")"
  E2E_VAULT="$E2E_ROOT/vault"
  mkdir -p "$E2E_VAULT"
  export AGENT_MEMO_VAULT="$E2E_VAULT"
  # Never inherit force-fs from caller for uncontrolled phases; suites set explicitly.
  unset MM_FORCE_FS 2>/dev/null || true
}

e2e_reset_vault() {
  rm -rf "$E2E_VAULT"
  mkdir -p "$E2E_VAULT/brain" "$E2E_VAULT/daily" "$E2E_VAULT/templates"
}

e2e_mm() {
  # Fresh process so MM_FORCE_FS / AGENT_MEMO_VAULT from environment apply at helper startup.
  AGENT_MEMO_VAULT="$E2E_VAULT" "$E2E_MM" "$@"
}

e2e_begin_fs_phase() {
  e2e_reset_vault
  export MM_FORCE_FS=1
  export AGENT_MEMO_VAULT="$E2E_VAULT"
}

e2e_begin_cli_phase() {
  e2e_reset_vault
  unset MM_FORCE_FS 2>/dev/null || true
  export MM_FORCE_FS=0
  export AGENT_MEMO_VAULT="$E2E_VAULT"
  e2e_register_vault || return 1
}

e2e_end_cli_phase() {
  e2e_unregister_vault || true
}

e2e_teardown() {
  e2e_end_cli_phase
  if [ "${E2E_KEEP:-0}" = 1 ]; then
    printf 'E2E_ROOT=%s\n' "$E2E_ROOT"
    printf 'E2E_VAULT=%s\n' "$E2E_VAULT"
  else
    [ -n "${E2E_ROOT:-}" ] && rm -rf "$E2E_ROOT"
  fi
}
```

注意：`e2e_resolve_repo` 里删掉草稿中重复的 `here=` 赋值，最终只保留从 `lib/` 上溯两级到 repo root 的那条路径。

- [ ] **Step 2: 写 `register.sh`**

```bash
#!/usr/bin/env bash
# scripts/e2e/lib/register.sh — ephemeral Obsidian vault registration.
# Restores obsidian.json from a full-file backup. Requires jq.

e2e_obsidian_json() {
  local f
  for f in \
    "$HOME/Library/Application Support/obsidian/obsidian.json" \
    "$HOME/.config/obsidian/obsidian.json"; do
    [ -f "$f" ] && { printf '%s' "$f"; return 0; }
  done
  return 1
}

e2e_register_vault() {
  local f id ts tmp
  command -v jq >/dev/null 2>&1 || {
    printf 'e2e: jq required to register vault for cli phase\n' >&2
    return 1
  }
  f="$(e2e_obsidian_json)" || {
    printf 'e2e: obsidian.json not found; cannot enter cli mode\n' >&2
    return 1
  }
  E2E_OBSIDIAN_JSON="$f"
  E2E_OBSIDIAN_BAK="$E2E_ROOT/obsidian.json.bak"
  cp "$f" "$E2E_OBSIDIAN_BAK" || return 1
  id="memovault-e2e"
  ts="$(python3 -c 'import time;print(int(time.time()*1000))' 2>/dev/null || printf '%s000' "$(date +%s)")"
  tmp="$(mktemp)"
  # Replace any prior memovault-e2e id, then set path to this run's vault.
  jq --arg id "$id" --arg path "$E2E_VAULT" --argjson ts "$ts" \
    '(.vaults //={}) | .vaults = ((.vaults // {}) | with_entries(select(.key != $id))) | .vaults[$id] = {path:$path, ts:$ts, open:false}' \
    "$f" > "$tmp" || { rm -f "$tmp"; return 1; }
  # Fix: jq pipeline above may drop top-level keys. Prefer:
  # jq --arg id "$id" --arg path "$E2E_VAULT" --argjson ts "$ts"
  #   '(.vaults) |= (del(.[$id]) + {($id): {path:$path, ts:$ts, open:false}})'
  mv "$tmp" "$f"
  E2E_REGISTERED=1
  printf 'e2e: registered vault id=%s path=%s\n' "$id" "$E2E_VAULT" >&2
}

e2e_unregister_vault() {
  [ "${E2E_REGISTERED:-0}" = 1 ] || return 0
  [ -n "${E2E_OBSIDIAN_BAK:-}" ] && [ -f "$E2E_OBSIDIAN_BAK" ] || return 0
  [ -n "${E2E_OBSIDIAN_JSON:-}" ] || return 0
  cp "$E2E_OBSIDIAN_BAK" "$E2E_OBSIDIAN_JSON"
  E2E_REGISTERED=0
  printf 'e2e: restored obsidian.json from backup\n' >&2
}
```

  '(.vaults) |= ((. // {}) | del(.[$id]) | . + {($id): {path:$path, ts:$ts, open:false}})' \

- [ ] **Step 1: 写 `run.sh`**

```bash
#!/usr/bin/env bash
# scripts/e2e/run.sh - MemoVault dual-mode e2e entry.
set -uo pipefail

HERE="$(cd "$(dirname "$0")" && pwd)"
. "$HERE/lib/assert.sh"
. "$HERE/lib/env.sh"
. "$HERE/lib/register.sh"

E2E_KEEP=0
E2E_FS_ONLY=0
E2E_CLI_ONLY=0
E2E_REPO_OVERRIDE=""

usage() {
  cat <<'U'
Usage: scripts/e2e/run.sh [--keep] [--fs-only] [--cli-only] [--repo <path>]
  --keep       leave E2E_ROOT and print E2E_VAULT (for protocol follow-up)
  --fs-only    escape hatch (not official gate)
  --cli-only   escape hatch (not official gate)
  --repo PATH  override repository root
U
}

while [ $# -gt 0 ]; do
  case "$1" in
    --keep) E2E_KEEP=1; shift ;;
    --fs-only) E2E_FS_ONLY=1; shift ;;
    --cli-only) E2E_CLI_ONLY=1; shift ;;
    --repo) E2E_REPO_OVERRIDE="$2"; shift 2 ;;
    -h|--help) usage; exit 0 ;;
    *) printf 'e2e: unknown arg: %s\n' "$1" >&2; usage; exit 2 ;;
  esac
done

e2e_resolve_repo || exit 1
e2e_setup_vault || exit 1
trap 'e2e_teardown' EXIT

# Syntax gate
bash -n "$E2E_MM" || { e2e_fail "bash -n memovault.sh" "parse error"; }
for f in "$E2E_REPO"/scripts/lib/*.sh "$HERE"/lib/*.sh; do
  [ -f "$f" ] || continue
  bash -n "$f" || e2e_fail "bash -n $(basename "$f")" "parse error"
done
# suites may not exist yet during incremental land; run.sh should skip missing
# until Task 4–6 add them. Final form: always require suites 01-05.

run_suites() {
  local phase="$1" s
  for s in 01-preflight 02-capture 03-retrieve 04-graph 05-organize; do
    if [ -f "$HERE/suites/${s}.sh" ]; then
      # shellcheck disable=SC1090
      E2E_PHASE="$phase" . "$HERE/suites/${s}.sh"
    else
      e2e_fail "suite $s" "missing $HERE/suites/${s}.sh"
    fi
  done
}

FS_OK=0
CLI_OK=0

if [ "$E2E_CLI_ONLY" != 1 ]; then
  printf '\n--- FS phase ---\n'
  e2e_begin_fs_phase
  pass_before=$E2E_FAIL
  run_suites fs
  if [ "$E2E_FAIL" -eq "$pass_before" ]; then FS_OK=1; fi
  # Better: track per-phase fail delta
fi

if [ "$E2E_FS_ONLY" != 1 ]; then
  printf '\n--- CLI phase ---\n'
  if e2e_begin_cli_phase; then
    out="$(e2e_mm preflight 2>/dev/null || true)"
    case "$out" in
      mode=cli*)
        run_suites cli
        ;;
      *)
        e2e_fail "cli-preflight" "expected mode=cli, got: $out"
        ;;
    esac
    e2e_end_cli_phase
  else
    e2e_fail "cli-register" "could not register vault or start cli phase"
  fi
fi

e2e_summary
# Official gate: both phases required unless escape hatch used
rc=0
if [ "$E2E_FS_ONLY" != 1 ] && [ "$E2E_CLI_ONLY" != 1 ]; then
  # require zero fails overall (suites assert both phases into same counters)
  [ "$E2E_FAIL" -eq 0 ] || rc=1
elif [ "$E2E_FAIL" -ne 0 ]; then
  rc=1
fi
exit "$rc"
```
